"""Probe (evaluator, C04 EVAL_2) follow-ups of the claim-before-sweep race:
(1) the preempted runner DOES report late (attempt 1) -> billed once via record_lost_attempt?
(2) the same race through the DD-A63 (d) CPU claim path (_lease_for_cpu_runner) in a mixed fleet."""
import harness
import runner_sim as rs
from mbnas_contracts.schemas.jobs import JobResult, JobStatus
from mbnas_contracts.util.time import iso_after, utcnow_iso
from mbnas_core.db import repo
from mbnas_core.db.core import connect, tx
from mbnas_core.runner_api import auth, routes

CPU = {"device": "cpu", "device_uuid": None, "device_name": "CPU probe"}


def _expire(db, job_id):
    with tx(db):
        db.execute("UPDATE jobs SET started_at = ?, heartbeat_at = ?, lease_expires_at = ? WHERE id = ?",
                   (iso_after(utcnow_iso(), -20), iso_after(utcnow_iso(), -17), iso_after(utcnow_iso(), -1), job_id))


def _ledger(db, run_id):
    return [(c["attempt"], c["kind"], round(c["gpu_seconds"], 2), (c.get("meta") or {}).get("reason_code"))
            for c in repo.list_costs(db, run_id)]


def test_late_result_after_preemption(core, runner_sim, runner_sim2, base_proto, db):
    run_id, _ = harness.validate_and_run(core, base_proto())
    runner_sim.register()
    first = runner_sim.claim()
    runner_sim.start(first)
    _expire(db, first.job_id)
    runner_sim2.register()
    second = runner_sim2.claim()
    assert second is not None and second.attempt == 2
    late = JobResult(job_id=first.job_id, status=JobStatus.FAILED, attempt=1, reason_code="CHILD_EXIT_NO_RESULT",
                     outputs={"retryable": True})
    r = runner_sim.result(first, runner_sim.commit_body(first, late, started_at=iso_after(utcnow_iso(), -20)))
    print("LATE_RESULT", r.status_code, r.json().get("code"), "LEDGER", _ledger(db, run_id))


def test_cpu_path_claim_before_sweep(core, rp_settings, base_proto, tmp_path, sweeper, db, monkeypatch):
    calls = {"n": 0}
    real = routes._lease_for_cpu_runner

    def counting(*a, **k):
        calls["n"] += 1
        return real(*a, **k)

    monkeypatch.setattr(routes, "_lease_for_cpu_runner", counting)
    proto = base_proto()
    proto["console"]["training"] = {"epoch_target_total": 2, "seed": 13}
    r = core.post("/api/v1/training-runs", headers=harness.auth(), json={
        "source": {"kind": "baseline", "baseline_id": "yolo11n"}, "protocol": proto, "mode": "scratch",
        "validation_enabled": False, "idempotency_key": "c04-probe-cpu-race"})
    assert r.status_code == 200, r.text
    run_id = r.json()["run_id"]
    conn = connect(rp_settings, init=False)
    try:
        toks = {rid: auth.create_principal(conn, "runner", runner_id=rid)[1] for rid in ("GPU0", "CPU1", "CPU2")}
    finally:
        conn.close()
    rs.RunnerSim(core, toks["GPU0"], "GPU0", tmp_path / "g0").register(kinds=["EVALUATE"])  # mixed fleet
    c1 = rs.RunnerSim(core, toks["CPU1"], "CPU1", tmp_path / "c1")
    c2 = rs.RunnerSim(core, toks["CPU2"], "CPU2", tmp_path / "c2")
    c1.register(**CPU)
    c2.register(**CPU)
    first = c1.claim()
    assert first is not None and first.run_id == run_id
    c1.start(first)
    _expire(db, first.job_id)
    second = c2.claim()  # CPU path, before any sweep
    print("CPU_PATH_CALLS", calls["n"], "SECOND", None if second is None else second.attempt)
    print("SWEEP", sweeper(), "LEDGER", _ledger(db, run_id))
