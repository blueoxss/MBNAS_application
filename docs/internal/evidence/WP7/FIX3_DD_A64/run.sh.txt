#!/bin/bash
# usage: run.sh <label> <pytest args...>   (cwd = TARGET_ROOT, CPU only, isolated test root + basetemp in scratchpad)
S=/tmp/claude-1001/-home-mgjeong-Desktop-etri-hd/8da2e29b-07e2-48e6-bf6a-07622ebcfa6c/scratchpad/wp3/dd_a64
label=$1; shift
cd /home/mgjeong/Desktop/etri_hd/mbnas_application || exit 2
export CUDA_VISIBLE_DEVICES="" PYTHONDONTWRITEBYTECODE=1 PIP_CONFIG_FILE=/dev/null MBNAS_TEST_ROOT=$S/test_root
rm -rf "$S/pt_$label"
s=$(date +%s); echo "start $(date -u +%FT%TZ) cwd=$PWD label=$label args=$*" > "$S/$label.meta"
timeout 3000 runtime/venv/bin/python -B -m pytest "$@" -p no:cacheprovider --basetemp "$S/pt_$label" > "$S/$label.txt" 2>&1
rc=$?
echo "exit=$rc dur=$(( $(date +%s)-s ))s end=$(date -u +%FT%TZ)" >> "$S/$label.meta"
cat "$S/$label.meta"
exit $rc
